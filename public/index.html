Inspector de Base de Datos Relacional (`cafeteria.db`)

Visualización en tiempo real de las sentencias SQL `CREATE TABLE` y los registros almacenados

1. Esquema Relacional SQL (`CREATE TABLE` en `server.js`)

CREATE TABLE IF NOT EXISTS usuarios (
    id_usuario INTEGER PRIMARY KEY,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    nombre TEXT NOT NULL,
    rol TEXT NOT NULL,
    activo INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS productos (
    id_producto INTEGER PRIMARY KEY,
    codigo TEXT UNIQUE NOT NULL,
    nombre TEXT NOT NULL,
    categoria TEXT NOT NULL,
    descripcion TEXT,
    precio REAL NOT NULL CHECK(precio > 0),
    existencia INTEGER NOT NULL CHECK(existencia >= 0),
    activo INTEGER NOT NULL DEFAULT 1,
    imagen TEXT
);

CREATE TABLE IF NOT EXISTS pedidos (
    id_pedido INTEGER PRIMARY KEY,
    numero_pedido TEXT UNIQUE NOT NULL,
    fecha_hora TEXT NOT NULL,
    estado TEXT NOT NULL,
    total REAL NOT NULL,
    motivo_cancelacion TEXT,
    id_usuario INTEGER NOT NULL
);
